// AUTOMATIC AWNING CONTROL - YOUR_HOME
// Closes the awning for rain/wind/gusts (including short-term
// forecasts), reopens it to the previous position once the
// weather is stable again, respects manual interventions and
// low temperatures.
// ============================================================

// ===================== CONFIGURATION =====================
let LAT = "YOUR_LATITUDE";
let LON = "YOUR_LONGITUDE";

let TIMER_INTERVAL = 600000; // 10 minutes

let WIND_THRESHOLD = 35;       // km/h, current/forecast average wind
let GUST_THRESHOLD = 40;       // km/h, current/forecast gusts
let TEMP_THRESHOLD = 5;        // °C, below which the motor stays idle (manual only)
let RAIN_CODE_THRESHOLD = 3;   // weather_code > 3 => close

let REQUIRED_GOOD_READINGS = 2;   // consecutive good readings required before reopening
let MANUAL_OVERRIDE_HOURS = 2;    // duration automation stays suspended after a manual action

// ===================== STATE (loaded/saved via KVS) =====================
let badReadingsCount = 0;
let manualOverrideUntil = 0; // timestamp in ms
let savedPosition = null;    // awning position before it was closed due to bad weather

function loadManualOverride() {
  Shelly.call("KVS.Get", { key: "tenda_manual_override_until" }, function (result, error_code) {
    if (error_code === 0 && result && result.value) {
      manualOverrideUntil = Number(result.value);
    }
  });
}

function loadSavedPosition() {
  Shelly.call("KVS.Get", { key: "tenda_saved_position" }, function (result, error_code) {
    if (error_code === 0 && result && result.value !== undefined) {
      savedPosition = Number(result.value);
      print("Saved position restored from KVS: " + savedPosition);
    }
  });
}

function setManualOverride() {
  manualOverrideUntil = Date.now() + MANUAL_OVERRIDE_HOURS * 3600 * 1000;
  savedPosition = null; // the user has taken back control, don't force an old position
  Shelly.call("KVS.Set", { key: "tenda_manual_override_until", value: String(manualOverrideUntil) });
  Shelly.call("KVS.Delete", { key: "tenda_saved_position" });
  print("Manual intervention detected: automation suspended for " + MANUAL_OVERRIDE_HOURS + "h.");
}

// Detects movements not originated by this script (a source other than the script's own)
// NOTE: check the actual "source" values on your firmware by temporarily printing
// print(JSON.stringify(status)), then update this condition accordingly.
Shelly.addStatusHandler(function (status) {
  if (status.component === "cover:0" && status.delta && status.delta.source) {
    let src = status.delta.source;
    if (src !== "script" && src !== "SHOS" && src !== "timer") {
      setManualOverride();
    }
  }
});

// ===================== SCHEDULES =====================
function setSchedules(enable) {
  Shelly.call("Schedule.List", {}, function (result, error_code, error_message) {
    if (error_code === 0 && result.jobs) {
      for (let i = 0; i < result.jobs.length; i++) {
        Shelly.call("Schedule.Update", { id: result.jobs[i].id, enable: enable });
      }
      print("Scheduled jobs:", enable ? "ENABLED" : "DISABLED");
    } else {
      print("Error retrieving schedules: " + error_message);
    }
  });
}

function closeCover(reason) {
  let status = Shelly.getComponentStatus("cover:0");

  if (status && status.current_pos === 0) {
    print("Awning already closed, skipping command (detected reason: " + reason + ").");
    setSchedules(false);
    return;
  }

  // Save the current position only if it hasn't already been saved
  // (avoids overwriting it with subsequent closures during the same bad weather event)
  if (status && status.current_pos !== null && savedPosition === null) {
    savedPosition = status.current_pos;
    Shelly.call("KVS.Set", { key: "tenda_saved_position", value: String(savedPosition) });
    print("Position saved before closing: " + savedPosition);
  }

  print("Closing awning. Reason: " + reason);
  setSchedules(false);
  Shelly.call("Cover.GoToPosition", { id: 0, pos: 0 }, function (res, err_c, err_m) {
    if (err_c !== 0) print("Cover command error: " + err_m);
  });
  notify("Awning automatically closed: " + reason);
}

// ===================== WEATHER =====================
function checkWeather() {
  // NOTE: the manual override no longer blocks the whole check.
  // The safety closure for rain/wind always stays active;
  // only the automatic reopening is suspended during the override (see below).

  let url = "https://api.open-meteo.com/v1/forecast?latitude=" + LAT +
            "&longitude=" + LON +
            "&current=precipitation,weather_code,wind_speed_10m,wind_gusts_10m,temperature_2m" +
            "&minutely_15=precipitation,weather_code,wind_speed_10m,wind_gusts_10m" +
            "&forecast_minutely_15=2" + // next 30 min, in 15-min steps
            "&timezone=auto";

  print("Checking YOUR_HOME weather...");

  Shelly.call("HTTP.GET", { url: url }, function (result, error_code, error_message) {
    if (error_code !== 0) {
      closeCover("network/API error (" + error_message + ")");
      return;
    }

    try {
      let data = JSON.parse(result.body);
      let cur = data.current;
      let temp = cur.temperature_2m;

      // Below temperature threshold: motor stays idle, no automatic action
      if (temp < TEMP_THRESHOLD) {
        print("Temperature " + temp + "°C below threshold (" + TEMP_THRESHOLD + "°C). No automatic action.");
        return;
      }

      // --- Current data ---
      let rainNow = cur.precipitation > 0 || cur.weather_code > RAIN_CODE_THRESHOLD;
      let windNow = cur.wind_speed_10m >= WIND_THRESHOLD || cur.wind_gusts_10m >= GUST_THRESHOLD;

      // --- Forecast (next 15-30 min) ---
      let m15 = data.minutely_15;
      let rainSoon = false;
      let windSoon = false;
      if (m15 && m15.precipitation) {
        for (let i = 0; i < m15.precipitation.length; i++) {
          if (m15.precipitation[i] > 0 || m15.weather_code[i] > RAIN_CODE_THRESHOLD) rainSoon = true;
          if (m15.wind_speed_10m[i] >= WIND_THRESHOLD || m15.wind_gusts_10m[i] >= GUST_THRESHOLD) windSoon = true;
        }
      }

      print("Now: rain=" + rainNow + " wind=" + windNow +
            " | 30min forecast: rain=" + rainSoon + " wind=" + windSoon);

      if (rainNow || windNow) {
        badReadingsCount = REQUIRED_GOOD_READINGS;
        closeCover(rainNow ? "rain detected" : "strong wind/gusts");
      } else if (rainSoon || windSoon) {
        badReadingsCount = REQUIRED_GOOD_READINGS;
        closeCover(rainSoon ? "rain forecast soon" : "strong wind/gusts forecast soon");
      } else {
        badReadingsCount = Math.max(0, badReadingsCount - 1);
        if (badReadingsCount === 0) {
          if (Date.now() < manualOverrideUntil) {
            print("Weather OK, but manual override active: skipping automatic reopening.");
          } else {
            print("Weather stable and OK. Re-enabling schedules.");
            setSchedules(true);

            if (savedPosition !== null) {
              print("Restoring awning to previous position: " + savedPosition);
              Shelly.call("Cover.GoToPosition", { id: 0, pos: savedPosition }, function (res, err_c, err_m) {
                if (err_c !== 0) print("Error restoring position: " + err_m);
              });
              savedPosition = null;
              Shelly.call("KVS.Delete", { key: "tenda_saved_position" });
            }
          }
        } else {
          print("Weather OK, waiting for confirmation (" + badReadingsCount + " readings remaining).");
        }
      }
    } catch (e) {
      print("Error reading data: " + e.message);
    }
  });
}

// ===================== STARTUP =====================
loadManualOverride();
loadSavedPosition();
checkWeather();
Timer.set(TIMER_INTERVAL, true, checkWeather);
